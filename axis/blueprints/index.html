▸ 3D Viewer Integration Snippet

Drop this into your 3D viewer's sidebar to wire the component rows (01 PIERS → 15 HRV) to their blueprint sheets. Each click opens the sheet in a new window at a fixed size; drawings are standalone so nothing needs to be proxied.

// 1. Registry: component ID → blueprint sheet (relative path).
const BLUEPRINTS = {
  "01": "blueprints/S-101-piers.html",
  "02": "blueprints/S-201-joists.html",
  "03": "blueprints/A-201-subfloor.html",
  "04": "blueprints/S-301-steel.html",
  "05": "blueprints/A-301-sip-walls.html",
  "06": "blueprints/A-302-windows.html",
  "07": "blueprints/A-303-door.html",
  "08": "blueprints/A-304-roof.html",
  "09": "blueprints/A-305-parapet.html",
  "10": "blueprints/A-306-skirting.html",
  "11": "blueprints/A-401-interior.html",
  "12": "blueprints/P-501-fixtures.html",
  "13": "blueprints/A-402-stairs.html",
  "14": "blueprints/M-601-minisplit.html",
  "15": "blueprints/M-602-hrv.html",
};

// 2. Handler — call from the "CLICK TO ENLARGE" button.
function openBlueprint(componentId) {
  const href = BLUEPRINTS[componentId];
  if (!href) return;
  window.open(
    href,
    `blueprint_${componentId}`,                         // reuse window per sheet
    "width=1200,height=860,noopener,noreferrer"
  );
}

// 3. Wire the sidebar rows. Example markup per row:
//   <li data-component-id="03">03 SUBFLOOR</li>
document.querySelectorAll("[data-component-id]").forEach(el => {
  el.addEventListener("click", () => openBlueprint(el.dataset.componentId));
});

// 4. Thumbnail preview (the small rendered sheet in the info panel)
//    — use an <iframe> so the SVG auto-scales to the container:
//   <iframe class="blueprint-thumb" src="blueprints/A-201-subfloor.html"
//           loading="lazy" sandbox="allow-same-origin"></iframe>
//   .blueprint-thumb { width:100%; aspect-ratio:900/620; border:1px solid #333;
//                       pointer-events:none; transform-origin:top left; }

For the index gallery you're looking at right now, thumbnails are inline mini-SVGs (no iframes). For the live 3D viewer you can either continue that pattern or point an <iframe> at each sheet and let the browser scale it.
